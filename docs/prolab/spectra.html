135    def consistency(self,
136                    std_threshold=.1,
137                    measurements='all',
138                    groupby='id',
139                    remove_suspicious=False,
140                    plot_suspicious=False,
141                    recursive=False,
142                    legend=True,
143                    plot_path=None):
144        '''
145        Performs a consistency analysis of the curves.
146
147        Parameters
148        ----------
149        std_threshold : float, optional
150            Multiple measurements of the same station that have a standard
151            deviation greater than this value at any wavelength will be marked
152            as suspicious. The default is 0.1.
153        measurements : str, optional
154            Use `all` (default) to analyze all curves together, `sample` to
155            analyze only sample curves, or `ref` to analyze only
156            reference curves.
157        groupby : str or list, optional
158            Indicate the attribute or list of attributes to be used for
159            grouping the observations. The default is `id`.
160        remove_suspicious : bool, optional
161            Whether the suspicious curves should be removed; if `True`, a new
162            Spectra object is returned. The default is `False`.
163        plot_suspicious : bool, optional
164            Whether to plot the groups with suspicious measurements.
165            The default is `False`.
166        recursive : bool, optional
167            When `remove_suspicious=True`, will make the function run again
168            until no suspicious measurements are left, or until a single
169            measurement is left. The default is `False`.
170        legend : bool, optional
171            Whether to insert a legend when `plot_suspicious=True`.
172            The default is `True`.
173        plot_path : str, optinal
174            If provided and if `plot_suspicious=True`, the plots are saved at
175            this folder. The default is `None`.
176
177        Notes
178        -----
179        When `remove_suspicious=True`, each suspicious station' replicate is
180        compared with the median curve, and the replicate with the highest
181        absolute difference is removed. It means that only one of the curves is
182        removed at a time. The option `recursive=True` may be used, but make
183        sure to use a reasonable threshold, otherwise the spectra will be
184        dropped until only a single curve is left.
185
186        Returns
187        -------
188        prolab.Spectra
189        '''
190
191        # Analyze only samples...
192        if measurements == 'sample':
193            data = self.rdata.loc[self.is_sample].copy()
194
195        # ... or only references...
196        elif measurements == 'ref':
197            data = self.rdata.loc[self.is_blank].copy()
198
199        # ... or everything together
200        else:
201            data = self.rdata.copy()
202
203        # Spectral standard deviation
204        std = (data
205               .groupby(groupby)
206               .std(numeric_only=True)
207               .filter(regex='\d'))
208
209        # Check if any wavelength has std over the threshold
210        std_over = (std > std_threshold).sum(axis=1) > 0
211
212        # IDs with std over threshold
213        ids_over = std_over.loc[std_over].index.tolist()
214
215        # If suspicious stations exist
216        if len(ids_over) > 0:
217
218            # Iterate over curves
219            for s in ids_over:
220
221                # Get suspicious curves
222                suspicious_curves = (data
223                                     .set_index(groupby)
224                                     .sort_index()
225                                     .loc[s]
226                                     .filter(regex='\d'))
227
228                # Plot them if required
229                if plot_suspicious:
230                    fig, ax = plt.subplots(figsize=(5.75, 5.75/2), dpi=300)
231                    suspicious_curves.transpose().plot(legend=legend,
232                                                       lw=0.5, ax=ax)
233                    # Labels
234                    ax.set(xlabel='Wavelength (nm)', ylabel='Measurement',
235                           title=f'Suspicious curve: ID {s}')
236                    # Grid
237                    ax.grid(which='both', lw=.25, color='black')
238                    # Legend if required
239                    if legend:
240                        ax.legend(edgecolor='black', fancybox=False)
241                    # Config
242                    sns.despine()
243                    # Save if required
244                    if plot_path:
245                        plt.savefig(plot_path / f'suspicious_id_{s}.jpg')
246                    plt.show()
247
248            # Calculate the median curves of each ID
249            median = (data
250                      .groupby(groupby)
251                      .transform('median', numeric_only=True)
252                      .filter(regex='\d'))
253            median.columns = median.columns.map(int)
254
255            # Calculate mean absolute deviation (MAD) from the median curve
256            mad = pd.DataFrame(data
257                               .filter(items=self.wls)
258                               .sub(median)
259                               .abs()
260                               .sum(axis=1)
261                               .rename('mad'))
262
263            # Add info
264            mad = mad.join(data.filter(regex='\D'))
265
266            # Keys with highest MAD
267            mad_max = mad.groupby(groupby).idxmax()
268
269            # Intersection of keys with higher MAD, and IDs that are suspicious
270            self.suspicious = mad_max.loc[ids_over].mad.tolist()
271
272            # Remove suspicious data
273            if remove_suspicious:
274
275                # Drop suspect spectra
276                to_keep = self.rdata.index.difference(self.suspicious)
277                filtered = self.rdata.loc[to_keep]
278
279                # Print keys removed
280                print('--------------------------')
281                print('Suspicious curves removed:')
282                for c in self.suspicious: print(c)
283                print('--------------------------')
284
285                # Create new Spectra object
286                new = Spectra(filtered)
287
288                # Repeat if required
289                if recursive:
290                    return new.consistency(std_threshold, measurements,
291                                           groupby, remove_suspicious,
292                                           plot_suspicious, recursive,
293                                           legend, plot_path)
294
295                # Or just return the new object
296                else:
297                    return new
298
299            # In case suspicious are not to be removed
300            else:
301                return self
302
303        # If suspicious stations do not exist
304        else:
305            self.suspicious = None
306            print('--------------------')
307            print('No suspicious curves')
308            print('--------------------')
309            return self

315    def get_absorption(self,
316                       use_blank=True,
317                       blank=None,
318                       wl_null_point_correction=600,
319                       pathlength=.1):
320        '''
321        Retrieves absorption from absorbance curves.
322
323        Parameters
324        ----------
325
326        use_blank : bool, optional
327            Indicates whether the data must be corrected using a blank.
328            The default is `True`.
329
330        blank : pd.DataFrame, optional
331            Only used when `use_blank=True`.
332            If `None`, the mean blank measurements
333            within the data will be used.
334            If a single curve is provided (wavelengths in the columns),
335            it will be used in all measurements.
336            If multiple curves are provided (salinity curves), the algorithm
337            will select the blank closer to the sample at 685 nm, 
338            following the IOCCG protocol.
339            The default is `None`.
340
341        wl_null_point_correction : int or array-like, optional
342            Wavelength to be used at the null point correction (offset).
343            If an integer is passed, the absorption at this wavelength will
344            be used.
345            If a tuple with the limits of a spectral interval is
346            provided, the mean absorption within the range will be used.
347            If `None`, the null point correction won't be executed.
348            The default is `None`.
349
350        pathlength : float, optional
351            The instrument optical path length in meters. The default is `.1`.
352
353        Returns
354        -------
355        prolab.Spectra
356            The same object is returned.
357            A new attribute `absorption` is added inplace.
358        '''
359
360        #------------------------
361        # Prepare data and blank
362        #------------------------
363
364        # If the blank curves from the dataset are to be used
365        if use_blank and blank is None:
366
367            # Get mean reference curves
368            blank = (self
369                     .absorbance
370                     .loc[self.is_blank]
371                     .mean()
372                     .rename('blank')
373                     .to_frame()
374                     .transpose())
375
376        # If a blank curve (or curves) is provided
377        elif use_blank and blank is not None:
378
379            # Prepare the curve(s)
380            blank.columns = blank.columns.map(int)
381
382        # In case no blank is to be used (i.e., curves already corrected)
383        else:
384            blank = pd.DataFrame({wl: 0 for wl in self.wls}, index=['blank'])
385
386        # Min and max wavelengths available
387        wlmin = max(self.wls.min(), blank.columns.min())
388        wlmax = min(self.wls.max(), blank.columns.max())
389        wl_range = range(wlmin, wlmax + 1)
390
391        # Get mean sample curves
392        # They're grouped by id in case of multiple measurements
393        mean = (self
394                .rdata
395                .loc[self.is_sample]
396                .groupby('id')
397                .mean(numeric_only=True)
398                .filter(regex='\d'))
399        mean.columns = mean.columns.map(int)
400
401        # Filter wavelengths
402        self.raw_absorbance = mean[wl_range]
403        blank = blank[wl_range]
404
405        #-----------------------
406        # Correction with blank
407        #-----------------------
408
409        # For single blank curves
410        if len(blank) == 1:
411            self.absorbance = self.raw_absorbance.sub(blank.squeeze(), axis=1)
412
413        # For multiple blank curves
414        # It was designed to account for salinity curves, and the selection
415        # is performed by minimizing the difference at 685 nm
416        else:
417
418            print('The correction will use the salinity curves provided')
419
420            # Dictionary to receive corrected curves
421            corr_dict = {}
422
423            # Dictionaries to save reference curves and their names
424            blank_dict = {}
425            blank_sal_dict = {}
426
427            # Iterate over curves
428            for st, curve in self.raw_absorbance.iterrows():
429
430                # Select a reference curve
431                blank_sel = (
432                    abs(blank.loc[:, 685:].sub(curve.loc[685:], axis=1))
433                    .mean(axis=1)
434                    .sort_values()
435                    .index[0]
436                    )
437
438                # Subtract selected reference
439                corr_dict[st] = (curve - blank.loc[blank_sel])
440
441                # Save the selected reference curve
442                blank_dict[st] = blank.loc[blank_sel]
443                blank_sal_dict[st] = blank_sel
444
445            # Corrected data
446            self.absorbance = pd.DataFrame(corr_dict).transpose()
447
448            # Reference used
449            self.blank_curves = pd.DataFrame(blank_dict).transpose()
450            self.blank_salinity = pd.DataFrame(blank_sal_dict, index=[0])
451
452        #-----------------------
453        # Null point correction
454        #-----------------------
455
456        _check = True
457
458        # Single wavelength
459        if isinstance(wl_null_point_correction, (int, float)):
460
461            # Null point correction offsets
462            self.npc_offset = self.absorbance[wl_null_point_correction]
463            absorbance_off = self.absorbance.sub(self.npc_offset , axis=0)
464
465        # Mean within an interval
466        elif isinstance(wl_null_point_correction, (list, tuple, np.ndarray)):
467
468            # Initial and final wavelengths
469            wli, wlf = wl_null_point_correction
470
471            # Offsets
472            self.npc_offset = self.absorbance.loc[:, wli:wlf].mean(axis=1)
473            absorbance_off = self.absorbance.sub(self.npc_offset , axis=0)
474
475        # No correction
476        elif wl_null_point_correction is None:
477            absorbance_off = self.absorbance
478            _check = False
479
480        else:
481            raise ValueError('Set a proper value to '
482                             +'"wl_null_point_correction"')
483
484        # Check according to IOCCG recommendation
485        if _check and any(self.npc_offset > .0015):
486            print('WARNING: there are offsets > 0.0015 AU in the null ' +
487                  'point correction')
488
489        # Calculate absorption
490        self.absorption = 2.303 * absorbance_off / pathlength
491
492        return self

496    # Method: fit exponential curve
497    # -------------------------------------------------------------------------
498    def expfit(self, wl_ref=443, wl_range=(350, 500)):
499        '''
500        Fits exponential-decaying curves.
501
502        Parameters
503        ----------
504
505        wl_ref : int, optional
506            Wavelength of reference for the fitting. The default is 443.
507
508        wl_range : tuple, optional
509            Limits of the wavelength range to be used in the curve fitting.
510            The default is (350, 500).
511
512        Returns
513        -------
514        prolab.Spectra
515            The same object is returned.
516            A new attribute `fitted` is added inplace.
517        '''
518
519        # Exponential curve to fit
520        def exp_curve(wl, abs_ref, slope):
521            return abs_ref * np.exp(-slope * (wl - wl_ref))
522
523        # Initial guesses for abs_ref and slope
524        init_guess = (0.1, 0.02)
525
526        # Curves to be fitted (y)
527        if isinstance(wl_range, (list, tuple, np.ndarray)):
528            ydata = self.absorption.loc[:, range(wl_range[0], wl_range[1])]
529        else:
530            ydata = self.absorption
531
532        # Wavelengths (x)
533        x = ydata.columns.map(int).to_numpy()
534
535        # Dictionary to store data
536        fit_dict = {'id': [], f'a{wl_ref:.0f}': [], 'slope': []}
537
538        # Fit coefficients
539        for st, y in ydata.iterrows():
540
541            # Fit curve (least squares)
542            coeffs, _ = scipy.optimize.curve_fit(exp_curve, x, y, init_guess)
543
544            # Store data
545            fit_dict['id'].append(st)
546            fit_dict[f'a{wl_ref:.0f}'].append(coeffs[0])
547            fit_dict['slope'].append(coeffs[1])
548
549        # Final dataframe
550        self.fitted = pd.DataFrame(fit_dict).set_index('id')
551
552        return self

557    def part_absorption(self,
558                        vol_diameter,
559                        unit='absorbance',
560                        wl_offset=800,
561                        use_tau=True,
562                        trans_pattern='T',
563                        wl_range=None,
564                        plot_raw=False,
565                        plot_absorbance=False):
566        '''
567        Retrieves particulate absorption based on the
568        transmittance-reflectance method (Tassan & Ferrari, 1995).
569
570        Parameters
571        ----------
572        unit : str, optional
573            Indicates whether the measurements of transmittance and reflectance
574            are in `decimal` (e.g., 0.20) or `percentual` form.
575            The default is 'percentual'.
576        wl_offset : int, optional
577            At this wavelength, the transmittance of the sample is ensured to
578            be lower than or equal to the transmittance of the reference.
579            Also, at this wavelength the total and depigmented curves of a
580            given sample are supposed to be equal, and depigmented spectra may
581            be offset to ensure that. The default is 800.
582        use_tau : bool, optional
583            Indicates if the factor tau should be used in the calculation of
584            the absorbance. The default is `True`.
585        trans_pattern : str, optional
586            The pattern (within the attribute `rmode`) used to identify
587            transmittance spectra. The default is `T`.
588        wl_range : array-like, optional
589            If provided, the calculation are limited to this interval. Make
590            sure that it includes the wavelength set in `wl_offset`.
591            The default is None.
592        plot_raw : bool, optional
593            Whether to plot the raw curves after offset correction.
594            The default is `False`.
595        plot_absorbance : bool, optional
596            Whether to plot the absorbance curves. The default is `False`.
597
598        Returns
599        -------
600        abs_total : TYPE
601            DESCRIPTION.
602        abs_depig : TYPE
603            DESCRIPTION.
604        '''
605
606        #--------------
607        # Prepare data
608        #--------------
609
610        # Wavelengths
611        wls = self.wls
612
613        # Get data
614        if unit == 'absorbance':
615            rdata = self.rdata.set_index('id').filter(regex='\d')
616            # Convert to transmittance/reflectance
617            data = np.power(10, -rdata)
618        elif unit == 'percentual':
619            data[wls] /= 100
620
621        # Get filtered volume and filter clearance area
622        # vol = vol_diameter.volume
623        # area = np.pi * (vol_diameter.diameter **2) / 4
624
625        # Curtail wavelength range
626        if wl_range is not None:
627            data = data.loc[:, wl_range[0]:wl_range[1]]
628            wls = np.arange(wl_range[0], wl_range[1] + 1)
629
630        # Set filters
631        fref = self.is_blank.to_numpy()
632        fsample = self.is_sample.to_numpy()
633        ftotal = self.is_total.to_numpy()
634        ftrans = (self.rmode == trans_pattern).to_numpy()
635
636        # Apply the filters to separate data
637        trans_total = data.loc[fsample & ftrans & ftotal]
638        trans_depig = data.loc[fsample & ftrans & ~ftotal]
639        refle_total = data.loc[fsample & ~ftrans & ftotal]
640        refle_depig = data.loc[fsample & ~ftrans & ~ftotal]
641
642        # Get references
643        Tblk_total = data.loc[fref & ftrans & ftotal].median()
644        Rblk_total = data.loc[fref & ~ftrans & ftotal].median()
645        Tblk_depig = data.loc[fref & ftrans & ~ftotal].median()
646        Rblk_depig = data.loc[fref & ~ftrans & ~ftotal].median()
647
648        #---------------
649        # Apply offsets
650        #---------------
651
652        # Transmittance cannot exceed the blanks
653        trans_blank_offset = Tblk_total[wl_offset] - trans_total[wl_offset]
654        trans_blank_offset.loc[trans_blank_offset > 0] = 0
655
656        # Update transmittance
657        trans_total = trans_total.add(trans_blank_offset, axis=0)
658
659        # Total and depigmented curves must match at the reference wavelength
660        trans_offset = trans_total[wl_offset] - trans_depig[wl_offset]
661        refle_offset = refle_total[wl_offset] - refle_depig[wl_offset]
662
663        # Update tables
664        trans_depig = trans_depig.add(trans_offset, axis=0)
665        refle_depig = refle_depig.add(refle_offset, axis=0)
666
667        #-----------------
668        # Plot offset data
669        #-----------------
670
671        if plot_raw:
672
673            # Long data
674            plot_data = self.rdata.melt(
675                id_vars=self.rdata.filter(regex='\D').columns.tolist()
676                )
677            plot_data['variable'] = plot_data.variable.map(int)
678            plot_data['is_total'] = plot_data.is_total.replace(
679                {True: 'Total', False: 'Extracted'})
680            plot_data['is_trans'] = plot_data.is_trans.replace(
681                {True: 'Transmittance', False: 'Reflectance'})
682
683            # Styling
684            hue_pal = {'Transmittance': 'r', 'Reflectance': 'b'}
685            style_pal = {'Total': (None, None), 'Extracted': (3, 1)}
686
687            # Iterate over stations
688            for st in self.stations:
689
690                # Create figure
691                fig, ax = plt.subplots(figsize=(5.75, 5.75), dpi=100)
692
693                # Get data
694                _dt = plot_data.loc[(plot_data.id == st)]
695
696                # Plot sample curves
697                sns.lineplot(data=_dt, x='variable', y='value',
698                             palette=hue_pal,
699                             style='is_total',
700                             style_order=['Total', 'Extracted'],
701                             dashes=style_pal,
702                             hue='is_trans',
703                             hue_order=['Transmittance', 'Reflectance'],
704                             ax=ax)
705
706                # Plot blanks
707                sns.lineplot(Tblk_total.transpose(), color='r', alpha=.5)
708                sns.lineplot(Tblk_depig.transpose(), color='r', alpha=.5,
709                             ls='dashed')
710                sns.lineplot(Rblk_total.transpose(), color='b', alpha=.5)
711                sns.lineplot(Rblk_depig.transpose(), color='b', alpha=.5,
712                             ls='dashed')
713
714                # Pega handles e labels
715                handles, labels = ax.get_legend_handles_labels()
716                
717                # Cria legenda sem título
718                ax.legend([handles[i] for i in [1, 2, 4, 5]],
719                          [labels[i] for i in [1, 2, 4, 5]],
720                          fontsize=8, ncols=2)
721
722                # Config
723                ax.set(title=st, xlabel='Wavelength (nm)',
724                       ylabel='Transmittance | Reflectance')
725                sns.despine()
726                plt.show()
727
728
729        #-----------------
730        # Blank correction
731        #-----------------
732
733        # Apply correction (sample/reference)
734        Tt = trans_total.div(Tblk_total, axis=1).copy()
735        Td = trans_depig.div(Tblk_depig, axis=1).copy()
736        Rt = refle_total.div(Rblk_total, axis=1).copy()
737        Rd = refle_depig.div(Rblk_depig, axis=1).copy()
738
739        #----------------
740        # Tau calculation
741        #----------------
742
743        if use_tau:
744            # Optical depth of transmittance
745            odt_total = np.log10(1 / Tt)
746            odt_depig = np.log10(1 / Td)
747
748            # Quantity used to calculate tau
749            odts_total = odt_total.sub(.5 * odt_total[750], axis=0)
750            odts_depig = odt_depig.sub(.5 * odt_depig[750], axis=0)
751
752            # Correction factor tau
753            tau_total = 1.15 - 0.17 * odts_total
754            tau_total[(odts_total <= .02) | (odts_total >= .7)] = 1
755
756            tau_depig = 1.15 - 0.17 * odts_depig
757            tau_depig[(odts_depig <= .02) | (odts_depig >= .7)] = 1
758
759        else:
760            tau_total = 1
761            tau_depig = 1
762
763        #------------------------------------
764        # Correction and absorption retrieval
765        #------------------------------------
766
767        # Calculate absorbances
768        absorbance_total = ((1 - Tt + Rblk_total * (Tt - Rt)) /
769                             (1 + Rblk_total * Tt * tau_total))
770
771        absorbance_depig = ((1 - Td + Rblk_depig * (Td - Rd)) /
772                             (1 + Rblk_depig * Td * tau_depig))
773
774        if plot_absorbance:
775            # Create figure
776            fig, axes = plt.subplots(1, 2, figsize=(5.75, 5.75/3), dpi=300,
777                                     sharey=False, constrained_layout=False)
778
779            # Plot total
780            absorbance_total.transpose().plot(legend=False, lw=.2,
781                                               color='blue', ax=axes[0])
782            # Configure
783            axes[0].set(title='Total')
784
785            # Plot depigmented
786            absorbance_depig.transpose().plot(legend=False, lw=.2,
787                                               color='blue', ax=axes[1])
788            # Configure
789            axes[1].set(title='Depigmented')
790            sns.despine()
791            plt.show()
792
793        # Calculate optical densities
794        od_total = np.log10(1 / (1 - absorbance_total))
795        od_depig = np.log10(1 / (1 - absorbance_depig))
796
797        # # Calculate absorption
798        abs_total = np.log(10) * .719 * (od_total ** 1.2287) #/ (vol / area)
799        abs_depig = np.log(10) * .719 * (od_depig ** 1.2287) #/ (vol / area)
800
801        return abs_total, abs_depig

136    def consistency(self,
137                    std_threshold=.1,
138                    measurements='all',
139                    groupby='id',
140                    remove_suspicious=False,
141                    plot_suspicious=False,
142                    recursive=False,
143                    legend=True,
144                    plot_path=None):
145        '''
146        Performs a consistency analysis of the curves.
147
148        Parameters
149        ----------
150        std_threshold : float, optional
151            Multiple measurements of the same station that have a standard
152            deviation greater than this value at any wavelength will be marked
153            as suspicious. The default is 0.1.
154        measurements : str, optional
155            Use `all` (default) to analyze all curves together, `sample` to
156            analyze only sample curves, or `ref` to analyze only
157            reference curves.
158        groupby : str or list, optional
159            Indicate the attribute or list of attributes to be used for
160            grouping the observations. The default is `id`.
161        remove_suspicious : bool, optional
162            Whether the suspicious curves should be removed; if `True`, a new
163            Spectra object is returned. The default is `False`.
164        plot_suspicious : bool, optional
165            Whether to plot the groups with suspicious measurements.
166            The default is `False`.
167        recursive : bool, optional
168            When `remove_suspicious=True`, will make the function run again
169            until no suspicious measurements are left, or until a single
170            measurement is left. The default is `False`.
171        legend : bool, optional
172            Whether to insert a legend when `plot_suspicious=True`.
173            The default is `True`.
174        plot_path : str, optinal
175            If provided and if `plot_suspicious=True`, the plots are saved at
176            this folder. The default is `None`.
177
178        Notes
179        -----
180        When `remove_suspicious=True`, each suspicious station' replicate is
181        compared with the median curve, and the replicate with the highest
182        absolute difference is removed. It means that only one of the curves is
183        removed at a time. The option `recursive=True` may be used, but make
184        sure to use a reasonable threshold, otherwise the spectra will be
185        dropped until only a single curve is left.
186
187        Returns
188        -------
189        prolab.Spectra
190        '''
191
192        # Analyze only samples...
193        if measurements == 'sample':
194            data = self.rdata.loc[self.is_sample].copy()
195
196        # ... or only references...
197        elif measurements == 'ref':
198            data = self.rdata.loc[self.is_blank].copy()
199
200        # ... or everything together
201        else:
202            data = self.rdata.copy()
203
204        # Spectral standard deviation
205        std = (data
206               .groupby(groupby)
207               .std(numeric_only=True)
208               .filter(regex='\d'))
209
210        # Check if any wavelength has std over the threshold
211        std_over = (std > std_threshold).sum(axis=1) > 0
212
213        # IDs with std over threshold
214        ids_over = std_over.loc[std_over].index.tolist()
215
216        # If suspicious stations exist
217        if len(ids_over) > 0:
218
219            # Iterate over curves
220            for s in ids_over:
221
222                # Get suspicious curves
223                suspicious_curves = (data
224                                     .set_index(groupby)
225                                     .sort_index()
226                                     .loc[s]
227                                     .filter(regex='\d'))
228
229                # Plot them if required
230                if plot_suspicious:
231                    fig, ax = plt.subplots(figsize=(5.75, 5.75/2), dpi=300)
232                    suspicious_curves.transpose().plot(legend=legend,
233                                                       lw=0.5, ax=ax)
234                    # Labels
235                    ax.set(xlabel='Wavelength (nm)', ylabel='Measurement',
236                           title=f'Suspicious curve: ID {s}')
237                    # Grid
238                    ax.grid(which='both', lw=.25, color='black')
239                    # Legend if required
240                    if legend:
241                        ax.legend(edgecolor='black', fancybox=False)
242                    # Config
243                    sns.despine()
244                    # Save if required
245                    if plot_path:
246                        plt.savefig(plot_path / f'suspicious_id_{s}.jpg')
247                    plt.show()
248
249            # Calculate the median curves of each ID
250            median = (data
251                      .groupby(groupby)
252                      .transform('median', numeric_only=True)
253                      .filter(regex='\d'))
254            median.columns = median.columns.map(int)
255
256            # Calculate mean absolute deviation (MAD) from the median curve
257            mad = pd.DataFrame(data
258                               .filter(items=self.wls)
259                               .sub(median)
260                               .abs()
261                               .sum(axis=1)
262                               .rename('mad'))
263
264            # Add info
265            mad = mad.join(data.filter(regex='\D'))
266
267            # Keys with highest MAD
268            mad_max = mad.groupby(groupby).idxmax()
269
270            # Intersection of keys with higher MAD, and IDs that are suspicious
271            self.suspicious = mad_max.loc[ids_over].mad.tolist()
272
273            # Remove suspicious data
274            if remove_suspicious:
275
276                # Drop suspect spectra
277                to_keep = self.rdata.index.difference(self.suspicious)
278                filtered = self.rdata.loc[to_keep]
279
280                # Print keys removed
281                print('--------------------------')
282                print('Suspicious curves removed:')
283                for c in self.suspicious: print(c)
284                print('--------------------------')
285
286                # Create new Spectra object
287                new = Spectra(filtered)
288
289                # Repeat if required
290                if recursive:
291                    return new.consistency(std_threshold, measurements,
292                                           groupby, remove_suspicious,
293                                           plot_suspicious, recursive,
294                                           legend, plot_path)
295
296                # Or just return the new object
297                else:
298                    return new
299
300            # In case suspicious are not to be removed
301            else:
302                return self
303
304        # If suspicious stations do not exist
305        else:
306            self.suspicious = None
307            print('--------------------')
308            print('No suspicious curves')
309            print('--------------------')
310            return self

316    def get_absorption(self,
317                       use_blank=True,
318                       blank=None,
319                       wl_null_point_correction=600,
320                       pathlength=.1):
321        '''
322        Retrieves absorption from absorbance curves.
323
324        Parameters
325        ----------
326
327        use_blank : bool, optional
328            Indicates whether the data must be corrected using a blank.
329            The default is `True`.
330
331        blank : pd.DataFrame, optional
332            Only used when `use_blank=True`.
333            If `None`, the mean blank measurements
334            within the data will be used.
335            If a single curve is provided (wavelengths in the columns),
336            it will be used in all measurements.
337            If multiple curves are provided (salinity curves), the algorithm
338            will select the blank closer to the sample at 685 nm, 
339            following the IOCCG protocol.
340            The default is `None`.
341
342        wl_null_point_correction : int or array-like, optional
343            Wavelength to be used at the null point correction (offset).
344            If an integer is passed, the absorption at this wavelength will
345            be used.
346            If a tuple with the limits of a spectral interval is
347            provided, the mean absorption within the range will be used.
348            If `None`, the null point correction won't be executed.
349            The default is `None`.
350
351        pathlength : float, optional
352            The instrument optical path length in meters. The default is `.1`.
353
354        Returns
355        -------
356        prolab.Spectra
357            The same object is returned.
358            A new attribute `absorption` is added inplace.
359        '''
360
361        #------------------------
362        # Prepare data and blank
363        #------------------------
364
365        # If the blank curves from the dataset are to be used
366        if use_blank and blank is None:
367
368            # Get mean reference curves
369            blank = (self
370                     .absorbance
371                     .loc[self.is_blank]
372                     .mean()
373                     .rename('blank')
374                     .to_frame()
375                     .transpose())
376
377        # If a blank curve (or curves) is provided
378        elif use_blank and blank is not None:
379
380            # Prepare the curve(s)
381            blank.columns = blank.columns.map(int)
382
383        # In case no blank is to be used (i.e., curves already corrected)
384        else:
385            blank = pd.DataFrame({wl: 0 for wl in self.wls}, index=['blank'])
386
387        # Min and max wavelengths available
388        wlmin = max(self.wls.min(), blank.columns.min())
389        wlmax = min(self.wls.max(), blank.columns.max())
390        wl_range = range(wlmin, wlmax + 1)
391
392        # Get mean sample curves
393        # They're grouped by id in case of multiple measurements
394        mean = (self
395                .rdata
396                .loc[self.is_sample]
397                .groupby('id')
398                .mean(numeric_only=True)
399                .filter(regex='\d'))
400        mean.columns = mean.columns.map(int)
401
402        # Filter wavelengths
403        self.raw_absorbance = mean[wl_range]
404        blank = blank[wl_range]
405
406        #-----------------------
407        # Correction with blank
408        #-----------------------
409
410        # For single blank curves
411        if len(blank) == 1:
412            self.absorbance = self.raw_absorbance.sub(blank.squeeze(), axis=1)
413
414        # For multiple blank curves
415        # It was designed to account for salinity curves, and the selection
416        # is performed by minimizing the difference at 685 nm
417        else:
418
419            print('The correction will use the salinity curves provided')
420
421            # Dictionary to receive corrected curves
422            corr_dict = {}
423
424            # Dictionaries to save reference curves and their names
425            blank_dict = {}
426            blank_sal_dict = {}
427
428            # Iterate over curves
429            for st, curve in self.raw_absorbance.iterrows():
430
431                # Select a reference curve
432                blank_sel = (
433                    abs(blank.loc[:, 685:].sub(curve.loc[685:], axis=1))
434                    .mean(axis=1)
435                    .sort_values()
436                    .index[0]
437                    )
438
439                # Subtract selected reference
440                corr_dict[st] = (curve - blank.loc[blank_sel])
441
442                # Save the selected reference curve
443                blank_dict[st] = blank.loc[blank_sel]
444                blank_sal_dict[st] = blank_sel
445
446            # Corrected data
447            self.absorbance = pd.DataFrame(corr_dict).transpose()
448
449            # Reference used
450            self.blank_curves = pd.DataFrame(blank_dict).transpose()
451            self.blank_salinity = pd.DataFrame(blank_sal_dict, index=[0])
452
453        #-----------------------
454        # Null point correction
455        #-----------------------
456
457        _check = True
458
459        # Single wavelength
460        if isinstance(wl_null_point_correction, (int, float)):
461
462            # Null point correction offsets
463            self.npc_offset = self.absorbance[wl_null_point_correction]
464            absorbance_off = self.absorbance.sub(self.npc_offset , axis=0)
465
466        # Mean within an interval
467        elif isinstance(wl_null_point_correction, (list, tuple, np.ndarray)):
468
469            # Initial and final wavelengths
470            wli, wlf = wl_null_point_correction
471
472            # Offsets
473            self.npc_offset = self.absorbance.loc[:, wli:wlf].mean(axis=1)
474            absorbance_off = self.absorbance.sub(self.npc_offset , axis=0)
475
476        # No correction
477        elif wl_null_point_correction is None:
478            absorbance_off = self.absorbance
479            _check = False
480
481        else:
482            raise ValueError('Set a proper value to '
483                             +'"wl_null_point_correction"')
484
485        # Check according to IOCCG recommendation
486        if _check and any(self.npc_offset > .0015):
487            print('WARNING: there are offsets > 0.0015 AU in the null ' +
488                  'point correction')
489
490        # Calculate absorption
491        self.absorption = 2.303 * absorbance_off / pathlength
492
493        return self

497    # Method: fit exponential curve
498    # -------------------------------------------------------------------------
499    def expfit(self, wl_ref=443, wl_range=(350, 500)):
500        '''
501        Fits exponential-decaying curves.
502
503        Parameters
504        ----------
505
506        wl_ref : int, optional
507            Wavelength of reference for the fitting. The default is 443.
508
509        wl_range : tuple, optional
510            Limits of the wavelength range to be used in the curve fitting.
511            The default is (350, 500).
512
513        Returns
514        -------
515        prolab.Spectra
516            The same object is returned.
517            A new attribute `fitted` is added inplace.
518        '''
519
520        # Exponential curve to fit
521        def exp_curve(wl, abs_ref, slope):
522            return abs_ref * np.exp(-slope * (wl - wl_ref))
523
524        # Initial guesses for abs_ref and slope
525        init_guess = (0.1, 0.02)
526
527        # Curves to be fitted (y)
528        if isinstance(wl_range, (list, tuple, np.ndarray)):
529            ydata = self.absorption.loc[:, range(wl_range[0], wl_range[1])]
530        else:
531            ydata = self.absorption
532
533        # Wavelengths (x)
534        x = ydata.columns.map(int).to_numpy()
535
536        # Dictionary to store data
537        fit_dict = {'id': [], f'a{wl_ref:.0f}': [], 'slope': []}
538
539        # Fit coefficients
540        for st, y in ydata.iterrows():
541
542            # Fit curve (least squares)
543            coeffs, _ = scipy.optimize.curve_fit(exp_curve, x, y, init_guess)
544
545            # Store data
546            fit_dict['id'].append(st)
547            fit_dict[f'a{wl_ref:.0f}'].append(coeffs[0])
548            fit_dict['slope'].append(coeffs[1])
549
550        # Final dataframe
551        self.fitted = pd.DataFrame(fit_dict).set_index('id')
552
553        return self

558    def part_absorption(self,
559                        vol_diameter,
560                        unit='absorbance',
561                        wl_offset=800,
562                        use_tau=True,
563                        trans_pattern='T',
564                        wl_range=None,
565                        plot_raw=False,
566                        plot_absorbance=False):
567        '''
568        Retrieves particulate absorption based on the
569        transmittance-reflectance method (Tassan & Ferrari, 1995).
570
571        Parameters
572        ----------
573        unit : str, optional
574            Indicates whether the measurements of transmittance and reflectance
575            are in `decimal` (e.g., 0.20) or `percentual` form.
576            The default is 'percentual'.
577        wl_offset : int, optional
578            At this wavelength, the transmittance of the sample is ensured to
579            be lower than or equal to the transmittance of the reference.
580            Also, at this wavelength the total and depigmented curves of a
581            given sample are supposed to be equal, and depigmented spectra may
582            be offset to ensure that. The default is 800.
583        use_tau : bool, optional
584            Indicates if the factor tau should be used in the calculation of
585            the absorbance. The default is `True`.
586        trans_pattern : str, optional
587            The pattern (within the attribute `rmode`) used to identify
588            transmittance spectra. The default is `T`.
589        wl_range : array-like, optional
590            If provided, the calculation are limited to this interval. Make
591            sure that it includes the wavelength set in `wl_offset`.
592            The default is None.
593        plot_raw : bool, optional
594            Whether to plot the raw curves after offset correction.
595            The default is `False`.
596        plot_absorbance : bool, optional
597            Whether to plot the absorbance curves. The default is `False`.
598
599        Returns
600        -------
601        abs_total : TYPE
602            DESCRIPTION.
603        abs_depig : TYPE
604            DESCRIPTION.
605        '''
606
607        #--------------
608        # Prepare data
609        #--------------
610
611        # Wavelengths
612        wls = self.wls
613
614        # Get data
615        if unit == 'absorbance':
616            rdata = self.rdata.set_index('id').filter(regex='\d')
617            # Convert to transmittance/reflectance
618            data = np.power(10, -rdata)
619        elif unit == 'percentual':
620            data[wls] /= 100
621
622        # Get filtered volume and filter clearance area
623        # vol = vol_diameter.volume
624        # area = np.pi * (vol_diameter.diameter **2) / 4
625
626        # Curtail wavelength range
627        if wl_range is not None:
628            data = data.loc[:, wl_range[0]:wl_range[1]]
629            wls = np.arange(wl_range[0], wl_range[1] + 1)
630
631        # Set filters
632        fref = self.is_blank.to_numpy()
633        fsample = self.is_sample.to_numpy()
634        ftotal = self.is_total.to_numpy()
635        ftrans = (self.rmode == trans_pattern).to_numpy()
636
637        # Apply the filters to separate data
638        trans_total = data.loc[fsample & ftrans & ftotal]
639        trans_depig = data.loc[fsample & ftrans & ~ftotal]
640        refle_total = data.loc[fsample & ~ftrans & ftotal]
641        refle_depig = data.loc[fsample & ~ftrans & ~ftotal]
642
643        # Get references
644        Tblk_total = data.loc[fref & ftrans & ftotal].median()
645        Rblk_total = data.loc[fref & ~ftrans & ftotal].median()
646        Tblk_depig = data.loc[fref & ftrans & ~ftotal].median()
647        Rblk_depig = data.loc[fref & ~ftrans & ~ftotal].median()
648
649        #---------------
650        # Apply offsets
651        #---------------
652
653        # Transmittance cannot exceed the blanks
654        trans_blank_offset = Tblk_total[wl_offset] - trans_total[wl_offset]
655        trans_blank_offset.loc[trans_blank_offset > 0] = 0
656
657        # Update transmittance
658        trans_total = trans_total.add(trans_blank_offset, axis=0)
659
660        # Total and depigmented curves must match at the reference wavelength
661        trans_offset = trans_total[wl_offset] - trans_depig[wl_offset]
662        refle_offset = refle_total[wl_offset] - refle_depig[wl_offset]
663
664        # Update tables
665        trans_depig = trans_depig.add(trans_offset, axis=0)
666        refle_depig = refle_depig.add(refle_offset, axis=0)
667
668        #-----------------
669        # Plot offset data
670        #-----------------
671
672        if plot_raw:
673
674            # Long data
675            plot_data = self.rdata.melt(
676                id_vars=self.rdata.filter(regex='\D').columns.tolist()
677                )
678            plot_data['variable'] = plot_data.variable.map(int)
679            plot_data['is_total'] = plot_data.is_total.replace(
680                {True: 'Total', False: 'Extracted'})
681            plot_data['is_trans'] = plot_data.is_trans.replace(
682                {True: 'Transmittance', False: 'Reflectance'})
683
684            # Styling
685            hue_pal = {'Transmittance': 'r', 'Reflectance': 'b'}
686            style_pal = {'Total': (None, None), 'Extracted': (3, 1)}
687
688            # Iterate over stations
689            for st in self.stations:
690
691                # Create figure
692                fig, ax = plt.subplots(figsize=(5.75, 5.75), dpi=100)
693
694                # Get data
695                _dt = plot_data.loc[(plot_data.id == st)]
696
697                # Plot sample curves
698                sns.lineplot(data=_dt, x='variable', y='value',
699                             palette=hue_pal,
700                             style='is_total',
701                             style_order=['Total', 'Extracted'],
702                             dashes=style_pal,
703                             hue='is_trans',
704                             hue_order=['Transmittance', 'Reflectance'],
705                             ax=ax)
706
707                # Plot blanks

632        fref = self.is_blank.to_numpy()
633        fsample = self.is_sample.to_numpy()
634        ftotal = self.is_total.to_numpy()
635        ftrans = (self.rmode == trans_pattern).to_numpy()
636
637        # Apply the filters to separate data
638        trans_total = data.loc[fsample & ftrans & ftotal]
639        trans_depig = data.loc[fsample & ftrans & ~ftotal]
640        refle_total = data.loc[fsample & ~ftrans & ftotal]
641        refle_depig = data.loc[fsample & ~ftrans & ~ftotal]
642
643        # Get references
644        Tblk_total = data.loc[fref & ftrans & ftotal].median()
645        Rblk_total = data.loc[fref & ~ftrans & ftotal].median()
646        Tblk_depig = data.loc[fref & ftrans & ~ftotal].median()
647        Rblk_depig = data.loc[fref & ~ftrans & ~ftotal].median()
648
649        #---------------
650        # Apply offsets
651        #---------------
652
653        # Transmittance cannot exceed the blanks
654        trans_blank_offset = Tblk_total[wl_offset] - trans_total[wl_offset]
655        trans_blank_offset.loc[trans_blank_offset > 0] = 0
656
657        # Update transmittance
658        trans_total = trans_total.add(trans_blank_offset, axis=0)
659
660        # Total and depigmented curves must match at the reference wavelength
661        trans_offset = trans_total[wl_offset] - trans_depig[wl_offset]
662        refle_offset = refle_total[wl_offset] - refle_depig[wl_offset]
663
664        # Update tables
665        trans_depig = trans_depig.add(trans_offset, axis=0)
666        refle_depig = refle_depig.add(refle_offset, axis=0)
667
668        #-----------------
669        # Plot offset data
670        #-----------------
671
672        if plot_raw:
673
674            # Long data
675            plot_data = self.rdata.melt(
676                id_vars=self.rdata.filter(regex='\D').columns.tolist()
677                )
678            plot_data['variable'] = plot_data.variable.map(int)
679            plot_data['is_total'] = plot_data.is_total.replace(
680                {True: 'Total', False: 'Extracted'})
681            plot_data['is_trans'] = plot_data.is_trans.replace(
682                {True: 'Transmittance', False: 'Reflectance'})
683
684            # Styling
685            hue_pal = {'Transmittance': 'r', 'Reflectance': 'b'}
686            style_pal = {'Total': (None, None), 'Extracted': (3, 1)}
687
688            # Iterate over stations
689            for st in self.stations:
690
691                # Create figure
692                fig, ax = plt.subplots(figsize=(5.75, 5.75), dpi=100)
693
694                # Get data
695                _dt = plot_data.loc[(plot_data.id == st)]
696
697                # Plot sample curves
698                sns.lineplot(data=_dt, x='variable', y='value',
699                             palette=hue_pal,
700                             style='is_total',
701                             style_order=['Total', 'Extracted'],
702                             dashes=style_pal,
703                             hue='is_trans',
704                             hue_order=['Transmittance', 'Reflectance'],
705                             ax=ax)
706
707                # Plot blanks
708                sns.lineplot(Tblk_total.transpose(), color='r', alpha=.5)
709                sns.lineplot(Tblk_depig.transpose(), color='r', alpha=.5,
710                             ls='dashed')
711                sns.lineplot(Rblk_total.transpose(), color='b', alpha=.5)
712                sns.lineplot(Rblk_depig.transpose(), color='b', alpha=.5,
713                             ls='dashed')
714
715                # Pega handles e labels
716                handles, labels = ax.get_legend_handles_labels()
717                
718                # Cria legenda sem título
719                ax.legend([handles[i] for i in [1, 2, 4, 5]],
720                          [labels[i] for i in [1, 2, 4, 5]],
721                          fontsize=8, ncols=2)
722
723                # Config
724                ax.set(title=st, xlabel='Wavelength (nm)',
725                       ylabel='Transmittance | Reflectance')
726                sns.despine()
727                plt.show()
728
729
730        #-----------------
731        # Blank correction
732        #-----------------
733
734        # Apply correction (sample/reference)
735        Tt = trans_total.div(Tblk_total, axis=1).copy()
736        Td = trans_depig.div(Tblk_depig, axis=1).copy()
737        Rt = refle_total.div(Rblk_total, axis=1).copy()
738        Rd = refle_depig.div(Rblk_depig, axis=1).copy()
739
740        #----------------
741        # Tau calculation
742        #----------------
743
744        if use_tau:
745            # Optical depth of transmittance
746            odt_total = np.log10(1 / Tt)
747            odt_depig = np.log10(1 / Td)
748
749            # Quantity used to calculate tau
750            odts_total = odt_total.sub(.5 * odt_total[750], axis=0)
751            odts_depig = odt_depig.sub(.5 * odt_depig[750], axis=0)
752
753            # Correction factor tau
754            tau_total = 1.15 - 0.17 * odts_total
755            tau_total[(odts_total <= .02) | (odts_total >= .7)] = 1
756
757            tau_depig = 1.15 - 0.17 * odts_depig
758            tau_depig[(odts_depig <= .02) | (odts_depig >= .7)] = 1
759
760        else:
761            tau_total = 1
762            tau_depig = 1
763
764        #------------------------------------
765        # Correction and absorption retrieval
766        #------------------------------------
767
768        # Calculate absorbances
769        absorbance_total = ((1 - Tt + Rblk_total * (Tt - Rt)) /
770                             (1 + Rblk_total * Tt * tau_total))
771
772        absorbance_depig = ((1 - Td + Rblk_depig * (Td - Rd)) /
773                             (1 + Rblk_depig * Td * tau_depig))
774
775        if plot_absorbance:
776            # Create figure
777            fig, axes = plt.subplots(1, 2, figsize=(5.75, 5.75/3), dpi=300,
778                                     sharey=False, constrained_layout=False)
779
780            # Plot total
781            absorbance_total.transpose().plot(legend=False, lw=.2,
782                                               color='blue', ax=axes[0])
783            # Configure
784            axes[0].set(title='Total')
785
786            # Plot depigmented
787            absorbance_depig.transpose().plot(legend=False, lw=.2,
788                                               color='blue', ax=axes[1])
789            # Configure
790            axes[1].set(title='Depigmented')
791            sns.despine()
792            plt.show()
793
794        # Calculate optical densities
795        od_total = np.log10(1 / (1 - absorbance_total))
796        od_depig = np.log10(1 / (1 - absorbance_depig))
797
798        # # Calculate absorption
799        abs_total = np.log(10) * .719 * (od_total ** 1.2287) #/ (vol / area)
800        abs_depig = np.log(10) * .719 * (od_depig ** 1.2287) #/ (vol / area)
801
802        return abs_total, abs_depig

Retrieves particulate absorption based on the transmittance-reflectance method (Tassan & Ferrari, 1995).

Parameters

The default is 'percentual'. wl_offset : int, optional At this wavelength, the transmittance of the sample is ensured to be lower than or equal to the transmittance of the reference. Also, at this wavelength the total and depigmented curves of a given sample are supposed to be equal, and depigmented spectra may be offset to ensure that. The default is 800. use_tau : bool, optional Indicates if the factor tau should be used in the calculation of the absorbance. The default is True. trans_pattern : str, optional The pattern (within the attribute rmode) used to identify transmittance spectra. The default is T. wl_range : array-like, optional If provided, the calculation are limited to this interval. Make sure that it includes the wavelength set in wl_offset. The default is None. plot_raw : bool, optional Whether to plot the raw curves after offset correction. The default is False. plot_absorbance : bool, optional Whether to plot the absorbance curves. The default is False.
